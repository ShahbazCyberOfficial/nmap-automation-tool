# Linux Installation Guide

# Update the package list
sudo apt update
# Install Git
sudo apt install git -y
# Clone the repository
sudo git clone https://github.com/ShahbazCyberOfficial/nmap-automation-tool.git

# Go to the project directory
cd nmap-automation-tool

# Make the script executable
sudo chmod +x nmap_automation.py

# Run the tool
sudo python3 nmap_automation.py

# This guide explains how to install and run the Nmap Automation Tool on Linux systems such as Kali Linux, Ubuntu, and Debian.
